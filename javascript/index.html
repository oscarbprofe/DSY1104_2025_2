💻 Ejemplos de Código

1. Selectores de Elementos

// Por ID
const elemento = document.getElementById('mi-id');

// Por selector CSS
const elemento = document.querySelector('.mi-clase');
const elementos = document.querySelectorAll('.mi-clase');

// Por clase (HTMLCollection)
const elementos = document.getElementsByClassName('mi-clase');

// Por etiqueta
const elementos = document.getElementsByTagName('div');

2. Manipulación de CSS

// Cambiar estilos directamente
elemento.style.color = 'red';
elemento.style.backgroundColor = 'blue';

// Agregar/quitar clases
elemento.classList.add('nueva-clase');
elemento.classList.remove('clase-antigua');
elemento.classList.toggle('activo');

// Verificar clase
if (elemento.classList.contains('activo')) {
    // hacer algo
}

3. Crear Elementos

// Crear elemento
const nuevoDiv = document.createElement('div');
nuevoDiv.textContent = 'Nuevo elemento';
nuevoDiv.className = 'elemento-dinamico';

// Agregar al DOM
contenedor.appendChild(nuevoDiv);

// Crear con innerHTML
contenedor.innerHTML = '<p>Nuevo párrafo</p>';

// Insertar en posición específica
contenedor.insertBefore(nuevoElemento, elementoReferencia);

4. LocalStorage

// Guardar datos
localStorage.setItem('clave', 'valor');
localStorage.setItem('usuario', JSON.stringify(objeto));

// Recuperar datos
const valor = localStorage.getItem('clave');
const objeto = JSON.parse(localStorage.getItem('usuario'));

// Eliminar datos
localStorage.removeItem('clave');
localStorage.clear(); // Eliminar todo
